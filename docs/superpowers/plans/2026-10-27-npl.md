> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `npl`, an English-language Textual terminal dashboard for public IP details, separated physical/VPN data usage, live throughput, speed tests, and local network information.

**Architecture:** Three focused Python modules: `collectors.py` runs and parses system/network commands, `formatting.py` contains deterministic aggregation/formatting, and `npl.py` owns the CLI and Textual UI. Existing `vnstat`, `ip`, `nmcli`, `iw`, `curl`, and `ping` provide the data; no daemon, root access, or extra database is added.

**Tech Stack:** Python 3.14, Textual 8.2.7, `unittest`, vnStat 2.12, Linux networking tools.

**Spec:** `docs/superpowers/specs/2026-10-27-npl-design.md`

- Work only inside `/home/t9/npl`, except the command wrapper at `/home/t9/.local/bin/npl`.

- This workspace is not a Git repository. Do not initialize Git and do not commit.
- Never require root, open a listening port, or install a daemon.
- Never sum physical and VPN usage. Display them separately.
- Use `subprocess` argument lists with `shell=False`; never interpolate data into a shell string.

- Missing data must render as `N/A` or `null`, never crash the dashboard.
- Speed-test confirmation is required in the interactive UI; `npl --speedtest` is the explicit CLI opt-in.

"""Pure formatting helpers for npl."""

from __future__ import annotations

from typing import Iterable

IEC_UNITS = ("B", "KiB", "MiB", "GiB", "TiB", "PiB")
SPARK_CHARS = "▁▂▃▄▅▆▇█"


def format_bytes(value: float | int | None) -> str:
    if value is None:
        return "N/A"
    number = max(float(value), 0.0)
    for unit in IEC_UNITS:
        if number < 1024.0 or unit == IEC_UNITS[-1]:
            if unit == "B":
                return f"{int(number)} B"
            return f"{number:.1f} {unit}"
        number /= 1024.0
    return "N/A"


def format_rate(value: float | None) -> str:
    if value is None:
        return "N/A"
    return f"{format_bytes(value)}/s"


def format_latency(value: float | None) -> str:
    if value is None:
        return "N/A"
    return f"{value:.0f} ms" if value >= 10 else f"{value:.1f} ms"


def format_budget(used: int, limit: int | None, width: int = 10) -> str:
    if not limit or limit <= 0:
        return "not set"
    ratio = min(max(used / limit, 0.0), 1.0)
    filled = round(ratio * width)
    bar = "█" * filled + "░" * (width - filled)
    return f"[{bar}] {ratio * 100:5.1f}%"


def sparkline(values: Iterable[float], width: int = 20) -> str:
    vals = [float(value) for value in values][-width:]
    if not vals:
        return ""
    low = min(vals)
    high = max(vals)
    if high <= 0:
        return SPARK_CHARS[0] * len(vals)
    if high == low:
        return SPARK_CHARS[len(SPARK_CHARS) // 2] * len(vals)
    span = high - low
    return "".join(
        SPARK_CHARS[round((value - low) / span * (len(SPARK_CHARS) - 1))]
        for value in vals
    )
```

from datetime import datetime, timedelta
from typing import Any, Sequence


def _window_sum(
    entries: Sequence[dict[str, Any]] | None,
    now: datetime,
    seconds: int,
) -> dict[str, int]:
    cutoff = now.timestamp() - seconds
    rx = 0
    tx = 0
    for entry in entries or []:
        timestamp = entry.get("timestamp")
        if timestamp is not None and timestamp >= cutoff:
            rx += int(entry.get("rx", 0) or 0)
            tx += int(entry.get("tx", 0) or 0)
    return {"rx": rx, "tx": tx}


def usage_windows(
    traffic: dict[str, Any],
    now: datetime,
) -> dict[str, dict[str, int]]:
    hours = traffic.get("hour", [])
    days = traffic.get("day", [])
    return {
        "last_24h": _window_sum(hours, now, 24 * 3600),
        "last_48h": _window_sum(hours, now, 48 * 3600),
        "last_7d": _window_sum(days, now, 7 * 86400),
        "last_30d": _window_sum(days, now, 30 * 86400),
    }


def available_days(
    days: Sequence[dict[str, Any]] | None,
    now: datetime,
) -> int:
    timestamps = [
        entry["timestamp"]
        for entry in (days or [])
        if entry.get("timestamp") is not None
    ]
    if not timestamps:
        return 0
    oldest = min(timestamps)
    span = int((now.timestamp() - oldest) // 86400) + 1
    return max(1, min(30, span))


def month_usage(
    traffic: dict[str, Any],
    now: datetime,
) -> dict[str, int]:
    rx = 0
    tx = 0
    for entry in traffic.get("month", []) or []:
        stamp = entry.get("timestamp")
        if stamp is None:
            continue
        entry_dt = datetime.fromtimestamp(stamp, tz=now.tzinfo)
        if entry_dt.year == now.year and entry_dt.month == now.month:
            rx += int(entry.get("rx", 0) or 0)
            tx += int(entry.get("tx", 0) or 0)
    return {"rx": rx, "tx": tx}


def daily_series(
    days: Sequence[dict[str, Any]] | None,
    now: datetime,
    count: int = 7,
) -> list[int]:
    buckets: dict[str, int] = {}
    for entry in days or []:
        stamp = entry.get("timestamp")
        if stamp is None:
            continue
        key = datetime.fromtimestamp(stamp, tz=now.tzinfo).date().isoformat()
        buckets[key] = (
            buckets.get(key, 0)
            + int(entry.get("rx", 0) or 0)
            + int(entry.get("tx", 0) or 0)
        )
    return [
        buckets.get((now.date() - timedelta(days=offset)).isoformat(), 0)
        for offset in range(count - 1, -1, -1)
    ]
```

"""System and network data collectors for npl."""

from __future__ import annotations

import re
import subprocess
from dataclasses import dataclass, field, replace
from typing import Any, Callable


@dataclass(frozen=True)
class CommandResult:
    stdout: str
    ok: bool
    error: str = ""


@dataclass
class NetworkInfo:
    hostname: str
    physical_interface: str | None = None
    vpn_interface: str | None = None
    gateway: str | None = None
    ipv4: str | None = None
    ipv6: str | None = None
    mac: str | None = None
    dns: list[str] = field(default_factory=list)
    ssid: str | None = None
    signal_dbm: int | None = None
    rx_bitrate_mbps: float | None = None
    tx_bitrate_mbps: float | None = None


Runner = Callable[..., CommandResult]


def run_command(
    args: list[str],
    timeout: float = 5.0,
    input_data: bytes | None = None,
) -> CommandResult:
    try:
        process = subprocess.run(
            args,
            input=input_data,
            capture_output=True,
            timeout=timeout,
            check=False,
        )
    except (OSError, subprocess.TimeoutExpired) as exc:
        return CommandResult("", False, str(exc))
    stdout = process.stdout.decode("utf-8", errors="replace")
    stderr = process.stderr.decode("utf-8", errors="replace")
    return CommandResult(
        stdout,
        process.returncode == 0,
        stderr or f"exit {process.returncode}",
    )


def _parse_nmcli_types(nmcli_text: str) -> dict[str, str]:
    types: dict[str, str] = {}
    for line in nmcli_text.splitlines():
        parts = line.split(":", 3)
        if len(parts) >= 2:
            types[parts[0]] = parts[1]
    return types


def _pick_address(link: dict[str, Any], family: str) -> str | None:
    candidates = [
        address
        for address in link.get("addr_info", [])
        if address.get("family") == family and address.get("scope") != "host"
    ]
    if not candidates:
        return None
    global_addresses = [
        address for address in candidates if address.get("scope") == "global"
    ]
    selected = (global_addresses or candidates)[0]
    return f"{selected['local']}/{selected['prefixlen']}"


def parse_network(
    address_json: list[dict[str, Any]],
    default_routes: list[dict[str, Any]],
    nmcli_text: str,
    hostname: str,
    preferred: str | None = None,
) -> NetworkInfo:
    links = {
        entry["ifname"]: entry for entry in address_json if "ifname" in entry
    }
    types = _parse_nmcli_types(nmcli_text)
    vpn_candidates = [
        name
        for name, interface_type in types.items()
        if interface_type in {"tun", "wireguard"} and name in links
    ]
    vpn_interface = vpn_candidates[0] if vpn_candidates else None

    route = None
    if preferred:
        route = next(
            (
                candidate
                for candidate in default_routes
                if candidate.get("dev") == preferred
            ),
            None,
        )
    if route is None:
        route = next(
            (
                candidate
                for candidate in default_routes
                if candidate.get("dev") in links
                and candidate.get("dev") != vpn_interface
            ),
            None,
        )
    if route is None and default_routes:
        route = default_routes[0]

    physical_interface = preferred or (route or {}).get("dev")
    if physical_interface not in links:
        physical_interface = next(
            (
                name
                for name, interface_type in types.items()
                if interface_type in {"wifi", "ethernet"} and name in links
            ),
            physical_interface,
        )
    if physical_interface not in links:
        return NetworkInfo(hostname=hostname, vpn_interface=vpn_interface)

    link = links[physical_interface]
    return NetworkInfo(
        hostname=hostname,
        physical_interface=physical_interface,
        vpn_interface=vpn_interface,
        gateway=(route or {}).get("gateway"),
        ipv4=_pick_address(link, "inet"),
        ipv6=_pick_address(link, "inet6"),
        mac=link.get("address"),
    )


def enrich_local(info: NetworkInfo, dns_text: str, iw_text: str) -> NetworkInfo:
    dns = [
        part.strip()
        for part in re.split(r"[|,]", dns_text)
        if part.strip()
    ]
    ssid_match = re.search(r"SSID:\s*(.+)", iw_text)
    signal_match = re.search(r"signal:\s*(-?\d+)\s*dBm", iw_text)
    rx_match = re.search(r"rx bitrate:\s*([\d.]+)\s*MBit/s", iw_text)
    tx_match = re.search(r"tx bitrate:\s*([\d.]+)\s*MBit/s", iw_text)
    return replace(
        info,
        dns=dns,
        ssid=ssid_match.group(1).strip() if ssid_match else None,
        signal_dbm=int(signal_match.group(1)) if signal_match else None,
        rx_bitrate_mbps=float(rx_match.group(1)) if rx_match else None,
        tx_bitrate_mbps=float(tx_match.group(1)) if tx_match else None,
    )


def _decode_json(result: CommandResult, fallback: Any) -> Any:
    if not result.ok:
        return fallback
    try:
        import json

        return json.loads(result.stdout)
    except (TypeError, ValueError):
        return fallback


def discover_network(
    preferred: str | None = None,
    runner: Runner = run_command,
) -> NetworkInfo:
    import socket

    address_json = _decode_json(
        runner(["ip", "-j", "address", "show"]),
        [],
    )
    default_routes = _decode_json(
        runner(["ip", "-j", "route", "show", "default"]),
        [],
    )
    nmcli = runner(
        ["nmcli", "-t", "-f", "DEVICE,TYPE,STATE,CONNECTION", "device", "status"]
    )
    nmcli_text = nmcli.stdout if nmcli.ok else ""
    info = parse_network(
        address_json,
        default_routes,
        nmcli_text,
        socket.gethostname(),
        preferred=preferred,
    )

    if info.physical_interface:
        dns_result = runner(
            ["nmcli", "-g", "IP4.DNS", "device", "show", info.physical_interface]
        )
        types = _parse_nmcli_types(nmcli_text)
        iw_result = (
            runner(["iw", "dev", info.physical_interface, "link"])
            if types.get(info.physical_interface) == "wifi"
            else CommandResult("", True, "")
        )

    if not org:
        return None, None
    match = re.match(r"^(AS\d+)\s+(.+)$", org)
    if match:
        return match.group(1), match.group(2)
    return None, org

        os.environ.get("XDG_CACHE_HOME", str(Path.home() / ".cache"))
    )

def fetch_ipinfo(
    family: str,
    runner: Runner = run_command,

        [
            "curl",

            "--max-time",

            "https://ipinfo.io/json",
        ],
        timeout=6,
    )

    if not isinstance(payload, dict) or not payload.get("ip"):
        return None

def _public_from_payload(
    payload: dict[str, Any],
    vpn_active: bool,
    stale_seconds: int,
) -> PublicIPInfo:
    ipv4 = payload.get("v4") or {}
    ipv6 = payload.get("v6") or {}

    asn, isp = _parse_org(primary.get("org"))

        ipv4=ipv4.get("ip"),
        ipv6=ipv6.get("ip"),
        isp=isp,
        asn=asn,
        city=primary.get("city"),
        region=primary.get("region"),
        country=primary.get("country"),
        vpn_active=vpn_active,
        stale_seconds=stale_seconds,
    )


def read_ip_info(
    vpn_active: bool,
    cache_path: Path | None = None,
    now: datetime | None = None,
    fetcher: Callable[[str], dict[str, Any] | None] = fetch_ipinfo,

def read_vnstat(
    interface: str | None,
    runner: Runner = run_command,
) -> dict[str, Any]:
    if not interface:
        return {}
    result = runner(["vnstat", "--json", "--iface", interface], timeout=8)
    payload = _decode_json(result, {})
    for item in payload.get("interfaces", []):
        if item.get("name") == interface:
            return item.get("traffic", {}) or {}
    return {}


def read_net_counters(
    interface: str | None,
    proc_path: Path = Path("/proc/net/dev"),
) -> tuple[int, int] | None:
    if not interface:
        return None
    try:
        lines = Path(proc_path).read_text(encoding="utf-8").splitlines()
    except OSError:
        return None
    for line in lines:
        if ":" not in line:
            continue
        name, values = line.split(":", 1)
        if name.strip() != interface:
            continue
        fields = values.split()
        if len(fields) >= 9:
            return int(fields[0]), int(fields[8])
    return None


def rates_from_counters(
    previous: tuple[int, int] | None,
    current: tuple[int, int] | None,
    elapsed: float,
) -> tuple[float, float]:
    if not previous or not current or elapsed <= 0:
        return 0.0, 0.0
    return (
        max(0.0, (current[0] - previous[0]) / elapsed),
        max(0.0, (current[1] - previous[1]) / elapsed),
    )


def ping_average(
    host: str | None,
    runner: Runner = run_command,
) -> float | None:
    if not host:
        return None
    result = runner(
        ["ping", "-n", "-c", "3", "-W", "1", host],
        timeout=6,
    )
    if not result.ok:
        return None
    times = [float(value) for value in re.findall(r"time=([\d.]+)\s*ms", result.stdout)]
    return sum(times) / len(times) if times else None


def _parse_curl_metric(output: str) -> tuple[float, int] | None:
    parts = output.strip().split()
    if len(parts) < 3:
        return None
    try:
        code = int(parts[0])
        speed = float(parts[1])
        size = int(float(parts[2]))
    except ValueError:
        return None
    if not 200 <= code < 300:
        return None
    return speed, size


def run_speedtest(
    down_bytes: int = 10_000_000,
    up_bytes: int = 2_000_000,
    runner: Runner = run_command,
    ping_fn: Callable[[str | None], float | None] = ping_average,
) -> dict[str, Any]:
    download_url = (
        f"https://speed.cloudflare.com/__down?bytes={int(down_bytes)}"
    )
    download = runner(
        [
            "curl",
            "-fsS",
            "-L",
            "--max-time",
            "30",
            "-o",
            "/dev/null",
            "-w",
            "%{http_code} %{speed_download} %{size_download}",
            download_url,
        ],
        timeout=35,
    )
    download_metric = _parse_curl_metric(download.stdout) if download.ok else None
    if download_metric is None:
        return {
            "download_bps": None,
            "upload_bps": None,
            "latency_ms": ping_fn("1.1.1.1"),
            "bytes_used": 0,
            "error": "download failed",
        }

    upload = runner(
        [
            "curl",
            "-fsS",
            "-L",
            "--max-time",
            "30",
            "-o",
            "/dev/null",
            "-H",
            "Content-Type: application/octet-stream",
            "--data-binary",
            "@-",
            "-w",
            "%{http_code} %{speed_upload} %{size_upload}",
            "https://speed.cloudflare.com/__up",
        ],
        timeout=35,
        input_data=b"\0" * int(up_bytes),
    )
    upload_metric = _parse_curl_metric(upload.stdout) if upload.ok else None
    if upload_metric is None:
        return {
            "download_bps": None,
            "upload_bps": None,
            "latency_ms": ping_fn("1.1.1.1"),
            "bytes_used": 0,
            "error": "upload failed",
        }

    return {
        "download_bps": download_metric[0] * 8,
        "upload_bps": upload_metric[0] * 8,
        "latency_ms": ping_fn("1.1.1.1"),
        "bytes_used": download_metric[1] + upload_metric[1],
        "error": None,
    }


def _valid_number(value: Any, default: float, minimum: float) -> float:
    try:
        parsed = float(value)
    except (TypeError, ValueError):
        return default
    return parsed if parsed >= minimum else default


def load_config(path: Path | None = None) -> tuple[dict[str, Any], str | None]:
    config = dict(DEFAULT_CONFIG)
    if path is None:
        base = Path(
            os.environ.get("XDG_CONFIG_HOME", str(Path.home() / ".config"))
        )
        path = base / "npl" / "config.toml"
    path = Path(path)
    if not path.exists():
        return config, None
    try:
        with path.open("rb") as handle:
            loaded = tomllib.load(handle)
    except (OSError, tomllib.TOMLDecodeError) as exc:
        return config, f"Config ignored: {exc}"
    if not isinstance(loaded, dict):
        return config, "Config ignored: top-level value must be a table"

    config["monthly_budget_gb"] = _valid_number(
        loaded.get("monthly_budget_gb", config["monthly_budget_gb"]),
        float(DEFAULT_CONFIG["monthly_budget_gb"]),
        0,
    )

        "ip_refresh_seconds",
        "usage_refresh_seconds",
        "local_refresh_seconds",
        "speedtest_down_bytes",
        "speedtest_up_bytes",
    ):
        config[key] = int(
            _valid_number(
                loaded.get(key, config[key]),
                float(DEFAULT_CONFIG[key]),
                1,
            )
        )
    for key in ("physical_interface", "vpn_interface"):
        value = loaded.get(key, config[key])
        config[key] = value if isinstance(value, str) else config[key]

def _speedtest_line(test: dict[str, Any] | None) -> str:
    if not test:

```python
#!/usr/bin/env python3
"""NPL terminal network dashboard."""

from __future__ import annotations

import argparse
import json
import sys
import time
from typing import Any

from textual.app import App, ComposeResult
from textual.widgets import Footer, Header, Static

from collectors import (
    collect_snapshot,
    load_config,
    rates_from_counters,
    read_net_counters,
    run_speedtest,
)
from formatting import render_dashboard

APP_VERSION = "0.1.0"


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="npl",
        description="Terminal network, usage, and speed dashboard",
    )
    parser.add_argument("--once", action="store_true", help="print one snapshot and exit")
    parser.add_argument("--json", action="store_true", help="print one JSON snapshot and exit")
    parser.add_argument("--speedtest", action="store_true", help="run a speed test and exit")
    parser.add_argument("--interface", help="override the physical interface")
    parser.add_argument(
        "--version",
        action="version",
        version=f"%(prog)s {APP_VERSION}",
    )
    return parser


class NplApp(App):
    CSS = """
    Screen { background: #101418; color: #d8dee9; }
    #dashboard-text { padding: 1 2; }
    """
    BINDINGS = [
        ("q", "quit", "Quit"),
        ("r", "refresh", "Refresh"),
        ("t", "speedtest", "Speed test"),
        ("i", "switch_interface", "Interface"),
        ("p", "toggle_pause", "Pause"),
        ("?", "help", "Help"),
    ]

    def __init__(self, snapshot: dict[str, Any], config: dict[str, Any]) -> None:
        super().__init__()
        self.snapshot = snapshot
        self.config = config
        self.samples: list[tuple[float, float]] = []
        self.previous_counters: tuple[int, int] | None = None
        self.previous_time: float | None = None
        self.paused = False
        self.speedtest_armed = False
        self.speedtest_running = False
        self.warning: str | None = None
        self.selected_interface = snapshot.get("local", {}).get("interface")

    def compose(self) -> ComposeResult:
        yield Header(show_clock=True)
        yield Static(id="dashboard-text", markup=False)
        yield Footer()

    def on_mount(self) -> None:
        self.set_interval(1.0, self.tick)
        refresh_seconds = min(
            float(self.config.get("usage_refresh_seconds", 60)),
            float(self.config.get("local_refresh_seconds", 30)),
            float(self.config.get("ip_refresh_seconds", 60)),
        )
        self.set_interval(
            refresh_seconds,
            self.action_refresh,
        )
        self.update_view()

    def update_view(self) -> None:
        content = render_dashboard(
            self.snapshot,
            self.samples,
            self.selected_interface,
            self.paused,
            self.warning,
        )
        self.query_one("#dashboard-text", Static).update(content)

    def tick(self) -> None:
        if self.paused:
            return
        now = time.monotonic()
        counters = read_net_counters(self.selected_interface)
        if counters and self.previous_counters and self.previous_time:
            rate = rates_from_counters(
                self.previous_counters,
                counters,
                now - self.previous_time,
            )
            self.samples.append(rate)
            self.samples = self.samples[-60:]
        self.previous_counters = counters
        self.previous_time = now
        self.update_view()

    def action_refresh(self) -> None:
        self.warning = "Refreshing..."
        self.update_view()
        self.run_worker(self._refresh_worker, thread=True, exclusive=True)

    def _refresh_worker(self) -> None:
        try:
            snapshot = collect_snapshot(self.config)
        except Exception as exc:
            self.call_from_thread(self._set_warning, f"Refresh failed: {exc}")
            return
        self.call_from_thread(self._apply_snapshot, snapshot)

    def _set_warning(self, message: str | None) -> None:
        self.warning = message
        self.update_view()

    def _apply_snapshot(self, snapshot: dict[str, Any]) -> None:
        existing_speedtest = self.snapshot.get("speedtest")
        self.snapshot = snapshot
        if existing_speedtest is not None:
            self.snapshot["speedtest"] = existing_speedtest
        interfaces = [
            self.snapshot.get("local", {}).get("interface"),
            self.snapshot.get("usage", {}).get("vpn", {}).get("interface"),
        ]
        if self.selected_interface not in [item for item in interfaces if item]:
            self.selected_interface = interfaces[0]
            self.previous_counters = None
            self.previous_time = None
        self.warning = None
        self.update_view()

    def action_switch_interface(self) -> None:
        interfaces = [
            self.snapshot.get("local", {}).get("interface"),
            self.snapshot.get("usage", {}).get("vpn", {}).get("interface"),
        ]
        interfaces = [item for item in interfaces if item]
        if not interfaces:
            return
        index = interfaces.index(self.selected_interface) if self.selected_interface in interfaces else -1
        self.selected_interface = interfaces[(index + 1) % len(interfaces)]
        self.samples.clear()
        self.previous_counters = None
        self.previous_time = None
        self.update_view()

    def action_toggle_pause(self) -> None:
        self.paused = not self.paused
        self.update_view()

    def action_help(self) -> None:
        self.warning = (
            "q Quit | r Refresh | t Speed test | i Switch interface | "
            "p Pause | ? Help"
        )
        self.update_view()

    def action_speedtest(self) -> None:
        if self.speedtest_running:
            return
        if not self.speedtest_armed:
            self.speedtest_armed = True
            self.warning = (
                "Speed test uses about 12 MB. Press t again within 10 seconds "
                "to confirm."
            )
            self.set_timer(10.0, self._disarm_speedtest)
            self.update_view()
            return
        self.speedtest_armed = False
        self.speedtest_running = True
        self.warning = "Running speed test..."
        self.update_view()
        self.run_worker(self._speedtest_worker, thread=True, exclusive=False)

    def _disarm_speedtest(self) -> None:
        if self.speedtest_armed:
            self.speedtest_armed = False
            self.warning = None
            self.update_view()

    def _speedtest_worker(self) -> None:
        try:
            result = run_speedtest(
                int(self.config["speedtest_down_bytes"]),
                int(self.config["speedtest_up_bytes"]),
            )
        except Exception as exc:
            result = {
                "download_bps": None,
                "upload_bps": None,
                "latency_ms": None,
                "bytes_used": 0,
                "error": str(exc),
            }
        self.call_from_thread(self._apply_speedtest, result)

    def _apply_speedtest(self, result: dict[str, Any]) -> None:
        self.speedtest_running = False
        self.snapshot["speedtest"] = result
        self.warning = f"Speed test failed: {result['error']}" if result.get("error") else None
        self.update_view()


def _print_snapshot(snapshot: dict[str, Any]) -> None:
    print(render_dashboard(snapshot))


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    config, config_warning = load_config()
    if args.interface:
        config["physical_interface"] = args.interface
    if config_warning:
        print(f"npl: {config_warning}", file=sys.stderr)

    if args.speedtest:
        result = run_speedtest(
            int(config["speedtest_down_bytes"]),
            int(config["speedtest_up_bytes"]),
        )
        print(json.dumps(result, indent=2, ensure_ascii=False))
        return 0 if result.get("error") is None else 1

    try:
        snapshot = collect_snapshot(
            config,
            interface=args.interface or config.get("physical_interface") or None,
        )
    except Exception as exc:
        print(f"npl: {exc}", file=sys.stderr)
        return 1

    if args.json:
        print(json.dumps(snapshot, indent=2, ensure_ascii=False))
        return 0
    if args.once:
        _print_snapshot(snapshot)
        return 0

    app = NplApp(snapshot, config)
    app.warning = config_warning
    app.run()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

The interactive dashboard keys are `q` quit, `r` refresh, `t` speed test,
`i` switch between physical and VPN throughput, `p` pause, and `?` help.

## Configuration

Copy `npl.toml.example` to `~/.config/npl/config.toml` and edit it. A
`monthly_budget_gb` of `0` disables the budget bar.

## Data Sources

- `vnstat --json` supplies 24/48-hour and 7/30-day usage.
- `/proc/net/dev` supplies live throughput.
- `ipinfo.io` supplies public IP metadata.
- Cloudflare speed endpoints supply the on-demand speed test.
- `ip`, `nmcli`, `iw`, and `ping` supply local network details.

Physical and VPN usage are always separate and never added together. The
monthly budget uses the physical interface because it best matches the
provider quota.

## Privacy and Safety

NPL requires no root privileges, starts no daemon, and opens no listening
port. It has no telemetry. It contacts only `ipinfo.io`, Cloudflare's
speed-test endpoints, and the configured gateway/`1.1.1.1` for latency.

## Limitations

- With a VPN active, public ISP and location describe the VPN exit node.
- Physical usage includes VPN overhead and some LAN traffic.
- `vnstat` history may be shorter than 30 days.
- A speed test consumes about 12 MB.
- Speed-test rates are reported in Mbit/s.
- Cloudflare's speed-test endpoints may be blocked on some networks.
